# ShortsAI — Python dependencies
# Install with:  pip install -r requirements.txt
# (Python 3.10+ recommended)

# Web server / API
fastapi>=0.110
uvicorn[standard]>=0.27       # ASGI server — run with: uvicorn app:app
python-multipart>=0.0.9       # REQUIRED by FastAPI for file uploads + form fields
pydantic>=2.5                 # (pulled in by FastAPI; pinned for clarity)

# Config
python-dotenv>=1.0            # loads GROQ_API_KEY / DEEPGRAM_API_KEY from .env

# AI / transcription APIs
groq>=0.11                    # Whisper transcription + translation/transliteration/titles
deepgram-sdk>=5,<8            # per-clip transcription (nova-3); v5+ API (listen.v1.media). 7.x no-arg
                             # DeepgramClient() no longer reads DEEPGRAM_API_KEY — key is passed explicitly in code.

# NOTE: Gemini needs NO package. It is the DEFAULT clip picker and is called over
# plain REST from the standard library, so setting GEMINI_API_KEY is genuinely all
# that is required. (It used to depend on google-generativeai, which meant a missing
# package silently disabled a perfectly good key — hence no SDK here on purpose.)

# Optional OFFLINE transcription fallback — never rate-limits / 500s. Recommended on
# the production GPU box (RTX 3050): set LOCAL_WHISPER_MODEL=medium, device auto-detects CUDA.
#   pip install faster-whisper        (plus a CUDA-enabled torch for GPU speed)
# faster-whisper>=1.0

# Video download
yt-dlp>=2024.4.9              # YouTube / generic URL downloads (invoked as a CLI)
gdown>=5.1                    # Google Drive link downloads

# Optional — used only to auto-scale parallel render workers by free RAM.
# If absent, the pipeline falls back to 2 workers. Safe to remove.
psutil>=5.9

# ─────────────────────────────────────────────────────────────
# NOT installable via pip — must be present on the system:
#   • ffmpeg + ffprobe   (cutting, scaling to 9:16, burning subtitles)
#   • A Devanagari font  (e.g. NotoSansDevanagari-Bold.ttf) for Hindi / dual
#                         captions — place it under ./fonts/  (see README)
# ─────────────────────────────────────────────────────────────

# ── Desktop window ────────────────────────────────────────────────────────────
# Runs the same local server inside a native window instead of a browser tab.
# It brings its own backend on Windows (pythonnet -> WebView2, already part of
# Edge) and macOS (pyobjc -> WKWebView). LINUX is the exception: the GTK bindings
# are a system package pip cannot supply —
#     Fedora        sudo dnf install python3-gobject webkit2gtk4.1
#     Debian/Ubuntu sudo apt install python3-gi gir1.2-webkit2-4.1
# Without a backend the app opens in the browser instead of failing, so this
# dependency is genuinely optional.
pywebview>=5,<7
